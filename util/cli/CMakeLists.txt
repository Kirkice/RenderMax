cmake_minimum_required(VERSION 3.10)
project(RenderMaxCLI LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 REQUIRED COMPONENTS Core)
get_filename_component(RENDERMAX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(rendermax-trace trace_backend.cpp "${RENDERMAX_ROOT}/qrenderdoc/Code/SystemTrace.cpp")
target_include_directories(rendermax-trace PRIVATE "${RENDERMAX_ROOT}/qrenderdoc")
target_link_libraries(rendermax-trace PRIVATE Qt5::Core)
if(MSVC)
  target_compile_options(rendermax-trace PRIVATE /utf-8)
endif()
install(TARGETS rendermax-trace RUNTIME DESTINATION bin)
install(PROGRAMS rendermax-cli rendermax_cli.py DESTINATION bin)
